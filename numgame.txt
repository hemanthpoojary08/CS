***number.js***import React from 'react';
import './number.css';
import { useState, useEffect } from 'react';

function Number() {
    const [initval, setInitval] = useState('');
    const [count, setCount] = useState(0);
    const [bgcol, setBgcol] = useState('');

    const handleChange = (e) => {
        setInitval(e.target.value);
        setCount(parseInt(e.target.value) || 0);
    };

    const handleInc = () => {
        setCount((prev) => prev + 1);
    };

    const handleDec = () => {
        setCount((prev) => prev - 1);
    };

    useEffect(() => {
        if (count % 10 === 0 && count != 0) {
            setBgcol('powderblue');
        } else {
            setBgcol('white');
        }
    }, [count]);

    return (
        <div className="game">
            <h1 style={{ backgroundColor: "black", color: "yellow" }}>Number Game</h1>
            <label htmlFor='textfield' className='textfield'>Initial Value: &nbsp;</label>
            <input type="text" id="textfield" className="textfield" value={initval} onChange={handleChange} /><br></br>
            <button className='bt' onClick={handleInc}>+</button>
            &nbsp; &nbsp;
            <button className="bt" onClick={handleDec}>-</button>
            <h2 className='countdisp'>
                Count:<span style={{ backgroundColor: bgcol, padding: "2px 6px" }}> {!isNaN(count) ? count : ''}</span>
            </h2>
        </div>
    );
}

export default Number;
